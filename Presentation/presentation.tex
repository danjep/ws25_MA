\begin{frame}{Introduction}
\begin{itemize}[<+->]
	\item The use of concurrent threads is very common
	\item Consider multiple concurrent threads that share memory
	\begin{itemize}
		\item They may interfere with each other's stored values
		\item Interleaving and reordering may create multiple executions with different results
	\end{itemize}
	\item Memory Consistency Models describe restrictions to how they can be executed
	\item It is important to verify and compare these models
\end{itemize}
\end{frame}

\begin{frame}{Introduction}
\vspace{0.5cm}
\begin{minipage}{0.7\textwidth}
\begin{itemize}[<+->]
	\item Programs may not be executed in the order they were written in
	\item Multiple Threads may get interleaved
	\item Different executions are possible
	\begin{itemize}
		\item Either thread is completely executed before the other ($a=0,b=1$ or $a=1,b=0$)
		\item Both write before the reads ($a=b=1$)
		\item Both reads before writes ($a=b=0$)
	\end{itemize}
\end{itemize}

\begin{frame}{Memory Consistency Models}
\begin{minipage}{0.7\textwidth}
\begin{itemize}
	\item Memory Consistency Models describe restrictions to the possible executions
	\item One Example is Sequential Consistency (\cite{Lam79}), where
	\begin{itemize}
		\item Operations are executed as if they were in some sequential order
		\item Operations of each thread follow the order in which they were written (program order (\po))
	\end{itemize}
	\item Forbids the $a=b=0$ execution
\end{itemize}

\begin{frame}{The Proof System: Removing the Constraints}
\begin{itemize}
	\item Every one of the constraints can be transformed into any other
	\begin{itemize}
	\item[] \begin{minipage}{0.7\textwidth}
	\vspace*{0.5cm}
	\begin{rules}{Constraint Transformation}
\inferrule*[left=($a\rightarrow i$)]{\texttt{irreflexive}~ \sf r^+}{\texttt{acyclic}~ \sf r} \and
\inferrule*[left=($e \rightarrow i$)]{\texttt{irreflexive}~ \sf r.r^{-1}}{\texttt{empty}~ \sf r} \and
\inferrule*[left=($i \rightarrow e$)]{\texttt{empty}~ \sf r \cap \id}{\texttt{irreflexive}~ \sf r} \and
\inferrule*[left=($e \rightarrow a$)]{\texttt{acyclic}~ \sf r\cup r^{-1}}{\texttt{empty}~ \sf r}
\end{rules}
	\end{minipage}  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Removing the Constraints}
\begin{itemize}
	\item Constraints can be eliminated when they are the same on both sides \\
	\begin{itemize}
		\item[] \begin{rules}{Constraint Elimination} 
\inferrule*[left=(empty)] {\sf r_1 \subseteq r_2}{\sf \texttt{empty}~ r_2 \subseteq \sf \texttt{empty}~ r_1} \and 
\inferrule*[left=(acyclic)] {\sf r_1 \subseteq r_2}{\sf \texttt{acyclic}~ r_2 \subseteq \texttt{acyclic}~ r_1} \and
\inferrule*[left={(irreflexive)}] {\sf r_1 \subseteq r_2}{\sf \texttt{irreflexive}~ r_2 \subseteq \texttt{irreflexive}~ r_1}
\end{rules}
	\end{itemize}
	\item Now we are left with an inclusion between relational terms $\relr \subseteq \Rels$
\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Transformations}
\begin{itemize}
	\item Transformations are rules based on equivalences and thus are invertible
	\item Examples are rules based on common operator properties like commutativity, distributivity and associativity
	\item Some Transformations are based on operator definitions
		\begin{mathpar} \footnotesize \inferrule*[left=($+$)]{\Omega[\sf r.r^*]}{\Omega[\sf r^+]} \and
\inferrule*[left=($?$)]{\Omega[\sf r \cup \id]}{\Omega[\sf r?]} \and
\inferrule*[left=($*$)]{\Omega[\sf \id \cup r^+]}{\Omega[\sf r^*]} \end{mathpar}
	\item Other Transformations are based on properties of relations
	\begin{mathpar} \footnotesize \inferrule*[left=($\0_{\cup}$)]{\Omega[\sf r]}{\Omega[\sf \0 \cup r]} \and
\inferrule*[left=($\0_{\cap}$)]{\Omega[\0]}{\Omega[\sf r \cap \0]} \and
\inferrule*[left=($\0_{conc}$)]{\Omega[\0]}{\Omega[\sf r.\0.s]} \end{mathpar}
\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Transformations}
\begin{itemize}
	\item {\fontfamily{lmr}\selectfont \textbf{Inversionlemma:} \textit{Let $\relr$ be a relational term that features an inversion. There is an expression $\relr'$ that only features inversions that are applied to relations that cannot be decomposed further (in CAT: base relations) with $\semsof{\relr'} = \semsof{\relr}$.}}
	\item Basically, Inversions can be assumed to only be used on base relations 
\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Transformations}
\begin{itemize}
	\item The proof of the lemma yields the rules for that transformation:
	\small \begin{rules}{Inversion}  \inferrule*[left=($-1_{\cup}$)]{\Omega[\sf r^{-1} \cup s^{-1}]}{\Omega[\sf (r \cup s)^{-1}]} \and
\inferrule*[left=($-1_{\cap}$)]{\Omega[\sf r^{-1} \cap s^{-1}]}{\Omega[\sf (r \cap s)^{-1}]} \and
\inferrule*[left=($-1_{.}$)]{\Omega[\sf s^{-1}.r^{-1}]}{\Omega[\sf (r.s)^{-1}]} \and
%\inferrule*[left=($-1_{\sim}$)]{\Omega[\sf \sim (r^{-1})]}{\Omega[\sf (\sim r)^{-1}]} \and nicht nötig wenn kein Komplement erlaubt wird
\inferrule*[left=($-1_{+}$)]{\Omega[\sf (r^{-1})^+]}{\Omega[\sf (r^+)^{-1}]} \and
\inferrule*[left=($-1_{*}$)]{\Omega[\sf (r^{-1})^*]}{\Omega[\sf (r^*)^{-1}]} \and
\inferrule*[left=($-1_{?}$)]{\Omega[\sf (r^{-1})?]}{\Omega[\sf (r?)^{-1}]} \and
\inferrule*[left=($\inv_\times$)]{\Omega[(B \times A)]}{\Omega[(A \times B)^\inv]} \end{rules}
\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Implications}
\begin{itemize}
	\item Implications are context-sensitive and not always invertible.
	\item Usually, we have two rules per operator, one for each side of the inclusion
	\begin{itemize}
		\item[] {\footnotesize \begin{rules}{Union}
%\inferrule*[left=($\cup_{2R}$)]{\it true}{\sf r \subseteq r \cup s} \and durch (U3R) und (ref)
\inferrule*[left=($\cup_{L}$)]{\relr \subseteq \relt \\ \rels \subseteq \relt}{\relr \cup \rels \subseteq \relt} \and
\inferrule*[left=($\cup_{R}$)]{\sf r \subseteq s}{\sf r \subseteq s \cup t} \and
\end{rules}}
	\end{itemize}
	\item This is where branching is possible
	\begin{itemize}
		\item Here, $\relr \cup \rels \subseteq \relt$ is true, if $\relr \subseteq \relt$ and $\rels \subseteq \relt$ are both true.
	\end{itemize}

\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Implications}
\begin{itemize}
	\item Moreover, we have axioms that are needed to close branches
	\begin{itemize}
		\item[] \footnotesize \begin{rules}{Axioms} \inferrule*[left=(ref)]{\it true}{\sf r \subseteq r} \and
\inferrule*[left=(empty)]{true}{\0 \subseteq \relr} \and
\inferrule*[left=(full)]{true}{\relr \subseteq (E \times E)} \and 
\inferrule*[left=($\cap_{L}$)]{\true}{\relr \cap \rels \subseteq \relr} \end{rules}
	\end{itemize}
	\item And rules for weakening and strengthening terms:
	\begin{itemize}
		\item[]\footnotesize \begin{rules}{Weak and Strength} \inferrule*[left=(strength)]{{\sf r \subseteq} \Delta[\sf s] \\ \sf s \subseteq t}{{\sf r \subseteq} \Delta[\sf t]} \and
\inferrule*[left=(weak)]{\Delta[\sf r] \subseteq s \\ \sf t \subseteq r}{\Delta[\sf t] \subseteq s} \end{rules}
	\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Implications}
\begin{itemize}
	\item For handling transitive closures on the left side of the inclusion, we use an invariant:
	\begin{itemize}
		\item[] \footnotesize \begin{rules}{Transitive Closure}
		\inferrule*[left=($+L$)]{\sf r \subseteq I \\ \sf r.I \subseteq I \\ \sf I \subseteq s}{\sf 		r^+ \subseteq s}
		\end{rules}
	\end{itemize}
	\item Invariant $\relI$ is a smallest fixed point for $\relr^+$
	\item The choice of the invariant is important, since not every inclusion can easily be resolved
\end{itemize}
\end{frame}

\begin{frame}{The Proof System: Assumptions}
\begin{itemize}
	\item The base relations of CAT have some inherent properties (e.g. \co{} is transitive)
	\item We capture this in a set of rules: 
	\vspace*{-0.7cm}\begin{mathpar}\hspace*{2cm} \small \inferrule*[left=($\co_{trans}$)]{true}{\co.\co \subseteq \co} \end{mathpar}

\end{itemize}

\section{Soundness}
\begin{frame}{Soundness}
\begin{itemize}
	\item For soundness, we show monotinicity of the unary and binary operators $f: (E \times E) \rightarrow (E \times E)$ and $g: (E\times E) \times (E \times E) \rightarrow (E \times E)$:
	\begin{itemize}
		\item For any $\Relr, \rels, \relt \subseteq (E \times E)$: $\relr \subseteq \Rels \impl f(\relr) \subseteq f(\rels)$ and $g(\relr,\relt) \subseteq g(\rels,\relt)$.
	\end{itemize}
	\item With monotonicity we prove the weakening and strengthening rules
	\item Subsequently, we prove every implication, monotonicity helps here too
	\begin{itemize}
		\item As an example, the proof for ($\cup_L$):
		\begin{eqnarray*}
\edgein \relr \cup \Rels &\impl & \edgein \relt \cup \rels \reason{\relr \subseteq \relt \text{ and binary monotonicity} }\\
&\impl & \edgein \relt \cup \relt \reason{\rels \subseteq \relt \text{ and binary monotonicity}} \\
&\impl & \edgein \relt
\end{eqnarray*}
	\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Soundness}
\begin{itemize}
	\item Only the proof for ($+_L$) is different:
	\begin{itemize}
		\item Induction for $\relr^+ \subseteq \relI$ on the number of concatenated $\relr$
		\item The premise $\relI \subseteq \rels$ then gives us the conclusion $\relr^+ \subseteq \Rels$
	\end{itemize}
	\item We expand the soundness for relational terms to MCMs
	\begin{itemize}
		\item Every MCM inclusion can be transformed into an relational inclusion
		\item The rules for constraint elimination and transformation guarantee this
	\end{itemize}
\end{itemize}
\end{frame}

\section{Completeness}
\begin{frame}{Completeness}
\begin{itemize}
	\item We show completeness by translating every proof $\pi$ of \cite{Gru+26} into our proof system
	\begin{itemize}
		\item For finite proofs a simple translation of the rules suffices
		\item For infinite proofs we present a procedure to make them finite
	\end{itemize}
	\item We assume the proofs to start with a relational inclusion, since we can expand these to MCMs as mentioned before
	\item Furthermore, we can assume the starting term in $\pi$ to be in the \textit{Relational Inclusion Standard Form} (RISF)
\end{itemize}
\end{frame}

\begin{frame}{Completeness}
\begin{itemize}
	\item The RISF is a logical formula with the following form:
	\[\phi_{\relr} \land \neg \phi_{\rels} = (e_1.\relr \cap e_2) \neq \emptyset, \neg ((e_1.\rels \cap e_2) \neq \emptyset)\]
	\item \cite{Gru+26} start with a logical formula including event constants
	\begin{itemize}
		\item The events mark the positions in the relational terms
		\item The formula as a whole describes a counterexample to the inclusion
		\item Their proof system applies rules which decompose the logical formulas until there is a contradiction
		\item The contradiction indicates that there exists no counterexample
	\end{itemize}
	
\end{itemize}
\end{frame}

\begin{frame}{Completeness}
\begin{minipage}{0.65\textwidth}
\begin{itemize}
	\item In $\pi$ a proof node consists of conjunct predicates, positive for the included term and negative for the including term 
	\item A counterexample would be an edge $\edgein \relr$ which is not in $\rels$.
	\item Branching occurs if there are multiple ways to obtain the counterexample
	\item Both branches need to end in contradictions
\end{itemize}
